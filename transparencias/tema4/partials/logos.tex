% Redefinir themecolor usando \csname para evitar problemas con @
\renewcommand{\themecolor}[1]{%
  \ifstrequal{#1}{colorbg}{%
    \setbeamercolor{background canvas}{bg=maincolor}%
    \def\@makelogo{t4translogo}%
    \setbeamercolor{normal text}{fg=white, bg=maincolor}%
    \setbeamercolor{structure}{fg=white}%
  }{%
    \setbeamercolor{background canvas}{bg=white}%
    \def\@makelogo{t4colorlogo}%
    \setbeamercolor{normal text}{fg=darkgray, bg=white}%
    \setbeamercolor{structure}{fg=maincolor}%
  }%
}
\makeatother
\themecolor{white}
